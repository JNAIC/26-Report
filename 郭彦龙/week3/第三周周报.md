```python train.py



\# Train the model

import torch

from torch import nn

from torch.utils.data import DataLoader          

from torchvision import datasets, transforms

from model import build\_model



epochs = 3

batch\_size = 64

learning\_rate = 0.01





def main():

&#x20;   torch.manual\_seed(42)

&#x20;   device = "cuda" if torch.cuda.is\_available() else "cpu"

&#x20;   print("device:", device, "| thread:", torch.get\_num\_threads())



&#x20;   tf = transforms.Compose(\[transforms.ToTensor(),

&#x20;                            transforms.Normalize((0.1307,), (0.3081,))])



&#x20;   train\_ds = datasets.MNIST("data", train=True, download=True, transform=tf)

&#x20;   test\_ds = datasets.MNIST("data", train=False, transform=tf)



&#x20;   train\_dl = DataLoader(train\_ds, batch\_size=batch\_size, shuffle=True, num\_workers=0)

&#x20;   test\_dl = DataLoader(test\_ds, batch\_size=256, num\_workers=0)      # 之前漏了这行



&#x20;   model = build\_model().to(device)   # 移动模型到 cpu 上



&#x20;   # 下面是优化器和损失函数

&#x20;   opt = torch.optim.Adam(model.parameters(), lr=learning\_rate)

&#x20;   loss\_fn = nn.CrossEntropyLoss()



&#x20;   for epoch in range(1, epochs + 1):

&#x20;       model.train()

&#x20;       total, correct, running = 0, 0, 0.0

&#x20;       for x, y in train\_dl:

&#x20;           x, y = x.to(device), y.to(device)

&#x20;           opt.zero\_grad()

&#x20;           loss = loss\_fn(model(x), y)

&#x20;           loss.backward()

&#x20;           opt.step()

&#x20;           running += loss.item() \* x.size(0)

&#x20;           correct += (model(x).argmax(1) == y).sum().item()

&#x20;           total += x.size(0)

&#x20;       print(f"epoch {epoch}  loss {running/total:.4f}  train\_acc {correct/total:.4f}")



&#x20;       # 评估部分

&#x20;       model.eval()

&#x20;       c = t = 0

&#x20;       with torch.no\_grad():

&#x20;           for x, y in test\_dl:

&#x20;               x, y = x.to(device), y.to(device)

&#x20;               c += (model(x).argmax(1) == y).sum().item()

&#x20;               t += y.size(0)

&#x20;       print(f"            test\_acc {c/t:.4f}")



&#x20;       # 保存部分，训练好的参数存成文件

&#x20;       torch.save(model.state\_dict(), f"checkpoints/mnist\_e{epoch}.pth")

&#x20;       print(f"            已保存 checkpoints/mnist\_e{epoch}.pth")





if \_\_name\_\_ == "\_\_main\_\_":        

&#x20;   main()

```
